##### new runs with baselines 

# Methods/baselines included in experiment/run_baseline.py:
# 1. proposed:
#    physics-informed testing model = reconstruction + latent uniform regularization,
#    with stability-aware asymmetric reconstruction loss.
# 2. regularization_only:
#    encoder-only baseline trained only with the latent uniform regularizer (no reconstruction model).
# 3. reconstruction_only:
#    sequence model trained only with reconstruction loss, with lambda_mmd = 0.
# 4. uniform_weight:
#    same sequence model as proposed, but with uniform reconstruction weighting
#    by setting asym_recon_delta = 1.
# 5. label_bce:
#    direct X -> Y classifier trained with BCE loss to predict the trajectory stability label.
#
# In the sweep:
# - proposed / regularization_only / reconstruction_only / uniform_weight
#   are compared by two-sample tests on their latent representations.
# - label_bce does not use a latent two-sample test; it reports the mean predicted
#   stability probability on the shifted test contexts.


for seed in $(seq 1 20); do

    --lambda-mmd 1e-2 \

done
